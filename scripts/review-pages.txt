async page => {
 const results=[];
 for (const width of [1440,390]) {
  await page.setViewportSize({width,height:900});
  for (const route of ['services','about','packages','careers','blog']) {
   await page.goto('http://localhost:5174/'+route);
   await page.getByRole('heading',{level:1}).waitFor();
   await page.evaluate(()=>document.fonts.ready);
   await page.emulateMedia({reducedMotion:'reduce'});
   await page.screenshot({path:'output/playwright/'+route+'-'+width+'.png',fullPage:true});
   results.push({route,width,overflow:await page.evaluate(()=>document.documentElement.scrollWidth>innerWidth),brokenImages:await page.locator('img').evaluateAll(imgs=>imgs.filter(i=>i.complete&&!i.naturalWidth).map(i=>i.src))});
  }
 }
 await page.goto('http://localhost:5174/');
 await page.screenshot({path:'output/playwright/home-mobile.png',fullPage:true});
 await page.setViewportSize({width:1440,height:900});
 await page.screenshot({path:'output/playwright/home-desktop.png',fullPage:true});
 console.log(JSON.stringify(results));
}
